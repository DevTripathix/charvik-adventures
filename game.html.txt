function updatePlayer(dt){
  const p=pl;
  const left=keys.ArrowLeft||keys.KeyA||virt.left, right=keys.ArrowRight||keys.KeyD||virt.right;

  /* ride platform */
  if(p.groundPlat){p.x+=p.groundPlat.dx;p.y+=p.groundPlat.dy;}

  /* dash overrides */
  if(p.dashT>0){
    p.dashT-=dt; p.vx=p.face*660; p.vy=0;
    ghosts.push({x:p.x,y:p.y,face:p.face,t:0});
    p.invuln=Math.max(p.invuln,.06);
  }else{
    const acc=p.grounded?ACC:AIRACC;
    if(left&&!right){p.vx-=acc*dt;p.face=-1;}
    else if(right&&!left){p.vx+=acc*dt;p.face=1;}
    else if(p.grounded){const f=FRIC*dt;if(Math.abs(p.vx)<=f)p.vx=0;else p.vx-=Math.sign(p.vx)*f;}
    p.vx=clamp(p.vx,-MOVE,MOVE);
    const holdJump=JUMPK.some(k=>keys[k]);
    const g=(p.vy<0&&holdJump)?GUP:GDN;
    p.vy=Math.min(p.vy+g*dt,MAXFALL);
  }

  /* jump logic */
  p.coyote=p.grounded?COYOTE:p.coyote-dt;
  p.jbuf-=dt; p.drop=Math.max(0,p.drop-dt);
  if(p.jbuf>0){
    if(p.coyote>0){                      /* ground jump */
      p.vy=-JUMPV;p.coyote=0;p.jbuf=0;p.grounded=false;p.sq=-.14;sfx.jump();
      burst(p.x+p.w/2,p.y+p.h,5,'#fff6e6',60);
    }else if(p.energy>=COST.wings){      /* spirit wings */
      p.vy=-545;p.jbuf=0;p.energy-=COST.wings;p.wingT=.35;sfx.wings();
      burst(p.x+p.w/2,p.y+p.h-6,8,'#8ee3c8',90);
    }
  }

  /* --- horizontal move & collide --- */
  p.x+=p.vx*dt;
  let ty0=Math.max(0,Math.floor(p.y/TILE)),ty1=Math.min(LH-1,Math.floor((p.y+p.h-1)/TILE));
  if(p.vx>0){const tx=Math.floor((p.x+p.w)/TILE);
    for(let ty=ty0;ty<=ty1;ty++)if(gi(tx,ty)===1){p.x=tx*TILE-p.w-.01;p.vx=0;break;}}
  else if(p.vx<0){const tx=Math.floor(p.x/TILE);
    for(let ty=ty0;ty<=ty1;ty++)if(gi(tx,ty)===1){p.x=(tx+1)*TILE+.01;p.vx=0;break;}}
  p.x=clamp(p.x,TILE,LPX-TILE-p.w);

  /* --- vertical move & collide --- */
  p.prevB=p.y+p.h;
  p.y+=p.vy*dt;
  const wasGround=p.grounded;
  p.grounded=false; p.groundPlat=null;
  const tx0=Math.floor((p.x+1)/TILE),tx1=Math.floor((p.x+p.w-1)/TILE);
  if(p.vy>=0){
    const ty=Math.floor((p.y+p.h)/TILE);
    for(let tx=tx0;tx<=tx1;tx++){const t=gi(tx,ty);
      if(t===1||(t===2&&p.prevB<=ty*TILE+10&&p.drop<=0)){
        p.y=ty*TILE-p.h;p.vy=0;p.grounded=true;break;}}
    for(const m of mplats){
      if(p.x+p.w>m.x&&p.x<m.x+m.w){
        const b=p.y+p.h;
        if(p.prevB<=m.y+12&&b>=m.y&&b<=m.y+m.h+16){
          p.y=m.y-p.h;p.vy=0;p.grounded=true;p.groundPlat=m;}}
    }
  }else{
    const ty=Math.floor(p.y/TILE);
    for(let tx=tx0;tx<=tx1;tx++)if(gi(tx,ty)===1){p.y=(ty+1)*TILE+.01;p.vy=0;break;}
  }
  if(p.grounded&&!wasGround){p.sq=.16;burst(p.x+p.w/2,p.y+p.h,4,'rgba(255,246,230,.8)',50);}
  p.sq*= (1-dt*8);

  /* hazards */
  const hx0=Math.floor((p.x+6)/TILE),hx1=Math.floor((p.x+p.w-6)/TILE);
  const hy0=Math.floor((p.y+8)/TILE),hy1=Math.floor((p.y+p.h-2)/TILE);
  let inLava=false,inSpike=false;
  for(let tx=hx0;tx<=hx1;tx++)for(let ty=hy0;ty<=hy1;ty++){
    const t=gi(tx,ty); if(t===3)inLava=true; if(t===4)inSpike=true;}
  if(inLava){hurt(true);return;}
  if(inSpike&&p.shieldT<=0)hurt(false);

  /* fell off the world */
  if(p.y>LH*TILE+60)hurt(true);

  /* timers & regen */
  p.invuln=Math.max(0,p.invuln-dt);
  p.fireCd=Math.max(0,p.fireCd-dt);
  p.dashCd=Math.max(0,p.dashCd-dt);
  p.shieldT=Math.max(0,p.shieldT-dt);
  p.shieldCd=Math.max(0,p.shieldCd-dt);
  p.wingT=Math.max(0,p.wingT-dt);
  p.energy=clamp(p.energy+4*dt,0,100);

  /* run dust */
  if(p.grounded&&Math.abs(p.vx)>180&&Math.random()<dt*14)
    parts.push({x:p.x+p.w/2-p.face*10,y:p.y+p.h-3,vx:-p.face*40,vy:-30,g:150,t:0,life:.4,sz:2.5,col:'rgba(255,246,230,.7)'});
}

  }
}

function updateEnemies(dt){
  const p=pl;
  for(const e of enemies){
    if(e.dead)continue;
    if(e.x<S.cam.x-700||e.x>S.cam.x+W+700)continue;    /* sleep far away */
    if(e.k==='crab'||e.k==='snail'){
      const sp=e.k==='crab'?42:24;
      e.x+=e.dir*sp*dt;
      const ax=e.dir>0?e.x+e.w+2:e.x-2;
      const foot=gi(Math.floor(ax/TILE),Math.floor((e.y+e.h+6)/TILE));
      const wall=gi(Math.floor(ax/TILE),Math.floor((e.y+e.h/2)/TILE));
      if(foot!==1&&foot!==2||wall===1||wall===2&&false)e.dir*=-1;
      if(foot===0)e.dir*=-1;
      if(wall===1)e.dir*=-1;
    }else if(e.k==='bird'){
      e.t+=dt*e.sp;
      e.x=e.ax+Math.sin(e.t*.8)*e.r;
      e.y=e.ay+Math.sin(e.t*2.3)*28;
      e.dir=Math.cos(e.t*.8)>0?1:-1;
    }else if(e.k==='slime'){
      if(e.ground){
        e.vx=0;e.wt-=dt;
        const px=p.x+p.w/2;
        if(Math.abs(px-(e.x+e.w/2))<300)e.dir=Math.sign(px-(e.x+e.w/2))||1;
        if(e.wt<=0){
          const ax=e.dir>0?e.x+e.w+4:e.x-4;
          const foot=gi(Math.floor(ax/TILE),Math.floor((e.y+e.h+6)/TILE));
          if(foot!==1){e.dir*=-1;}else{e.vy=-430;e.vx=e.dir*90;e.ground=false;}
          e.wt=.7+Math.random()*.6;
        }
      }else{
        e.vy+=GRAV*dt;e.x+=e.vx*dt;e.y+=e.vy*dt;
        if(e.vy>0&&solidAt(e.x+e.w/2,e.y+e.h)){e.y=Math.floor((e.y+e.h)/TILE)*TILE-e.h;e.vy=0;e.ground=true;e.vx=0;}
      }
    }
    /* touch player */
    if(S.mode!=='play')continue;
    if(p.x<e.x+e.w&&p.x+p.w>e.x&&p.y<e.y+e.h&&p.y+p.h>e.y){
      const stomp=p.vy>140&&p.prevB<=e.y+16;
      if(p.shieldT>0){killEnemy(e);continue;}
      if(stomp&&e.k!=='snail'){
        killEnemy(e);
        p.vy=JUMPK.some(k=>keys[k])?-560:-430;
        continue;
      }
      /* damage */
      if(p.invuln<=0&&p.shieldT<=0){
        p.invuln=1.4;p.hearts--;renderHearts();sfx.hurt();S.shake=Math.max(S.shake,8);
        burst(p.x+p.w/2,p.y+p.h/2,10,'#e2574c',160);
        if(p.hearts<=0){startDeath();}else{p.vy=-330;p.vx=Math.sign(p.x+p.w/2-(e.x+e.w/2))*260||-260;}
      }
    }
  }
  enemies=enemies.filter(e=>!e.dead);
}

function drawPlayer(){
  const p=pl;
  if(S.mode==='play'&&p.invuln>0&&p.dashT<=0&&Math.floor(S.time*16)%2===0)return;
  /* soft shadow */
  const gtx=Math.floor((p.x+p.w/2)/TILE);
  for(let ty=Math.floor((p.y+p.h)/TILE);ty<LH;ty++){
    if(gi(gtx,ty)===1){const d=ty*TILE-(p.y+p.h);
      if(d<180){ctx.globalAlpha=.16*(1-d/180);ctx.fillStyle='#000';
        ctx.beginPath();ctx.ellipse(p.x+p.w/2-S.cam.x,ty*TILE-S.cam.y+3,16,5,0,0,7);ctx.fill();ctx.globalAlpha=1;}break;}
  }
  /* dash ghosts */
  for(const g of ghosts){
    ctx.globalAlpha=.28*(1-g.t/.28);
    ctx.fillStyle='#8ee3c8';
    ctx.beginPath();ctx.roundRect(g.x-S.cam.x,g.y-S.cam.y,p.w,p.h,10);ctx.fill();
  }
  ctx.globalAlpha=1;
  ctx.save();
  ctx.translate(p.x+p.w/2-S.cam.x,p.y+p.h-S.cam.y);
  ctx.scale(p.face,1);
  ctx.scale(1+p.sq,1-p.sq);
  const run=Math.abs(p.vx)>20&&p.grounded, ph=S.time*11;
  const la=run?Math.sin(ph)*7:0, lb=run?-Math.sin(ph)*7:0;
  /* legs */
  ctx.fillStyle='#e8a06b';
  ctx.beginPath();ctx.roundRect(-8+la*.6,-16,6,16,3);ctx.fill();
  ctx.beginPath();ctx.roundRect(2+lb*.6,-16,6,16,3);ctx.fill();
  ctx.fillStyle='#4a3524';
  ctx.beginPath();ctx.roundRect(-9+la*.6,-4,8,5,2);ctx.fill();
  ctx.beginPath();ctx.roundRect(1+lb*.6,-4,8,5,2);ctx.fill();
  /* wings flash */
  if(p.wingT>0){
    const ws=Math.sin(p.wingT/.35*Math.PI);
    ctx.fillStyle='#8ee3c8';
    ctx.save();ctx.translate(-12,-34);ctx.rotate(-.5+ws*.4);ctx.beginPath();ctx.ellipse(-10,0,15,6,0,0,7);ctx.fill();ctx.restore();
    ctx.save();ctx.translate(-12,-30);ctx.rotate(.2-ws*.4);ctx.beginPath();ctx.ellipse(-10,2,13,5,0,0,7);ctx.fill();ctx.restore();
  }
  /* scarf */
  const sway=Math.sin(S.time*9)*3+(p.dashT>0?6:0);
  ctx.fillStyle='#e2574c';
  ctx.beginPath();ctx.moveTo(-2,-33);ctx.lineTo(-16-sway,-27);ctx.lineTo(-4,-26);ctx.closePath();ctx.fill();
  /* backpack */
  ctx.fillStyle='#8a5a33';ctx.beginPath();ctx.roundRect(-15,-33,8,13,3);ctx.fill();
  /* body */
  ctx.fillStyle='#2a9d8f';ctx.beginPath();ctx.roundRect(-10,-34,20,20,6);ctx.fill();
  ctx.fillStyle='#23857a';ctx.fillRect(-10,-18,20,3);
  /* head */
  ctx.fillStyle='#f0b183';ctx.beginPath();ctx.arc(1,-42,9,0,7);ctx.fill();
  ctx.fillStyle='#2b1d16';ctx.beginPath();ctx.arc(4.5,-43,1.7,0,7);ctx.fill();
  /* hat */
  ctx.fillStyle='#edc26b';
  ctx.beginPath();ctx.ellipse(1,-48,14.5,4.5,0,0,7);ctx.fill();
  ctx.beginPath();ctx.arc(1,-48,8.5,Math.PI,0);ctx.fill();
  ctx.fillStyle='#d64545';ctx.fillRect(-7.5,-51,17,3);
  ctx.restore();
  /* shield bubble */
  if(p.shieldT>0){
    const cx2=p.x+p.w/2-S.cam.x,cy2=p.y+p.h/2-S.cam.y,pu=.5+.5*Math.sin(S.time*6);

    ctx.beginPath();ctx.arc(cx2,cy2,27,S.time*3+3.14,S.time*3+4.24);ctx.stroke();
  }
}

let lastT=performance.now();
function loop(now){
  requestAnimationFrame(loop);
  let dt=Math.min((now-lastT)/1000,.033);lastT=now;
  S.time+=dt;
  if(S.mode==='menu'){
    S.cam.x=260+(0.5+0.5*Math.sin(S.time*.07))*560;
    S.cam.y=LH*TILE-H;
    updateFx(dt);
  }else if(S.mode==='play'){
    S.playT+=dt;
    updateMplats(dt);
    updatePlayer(dt);
    if(S.mode==='play'){
      updateEnemies(dt);
      updateProjs(dt);
      updatePickups(dt);
      updateCam(dt);
      syncTimerOnly();
    }
    updateFx(dt);
    syncHUD();
  }else if(S.mode==='dying'){
    S.dieT+=dt;
    pl.vy+=GRAV*dt;pl.y+=pl.vy*dt;pl.x+=pl.vx*dt*.4;
    if(S.dieT>1.3){
      S.mode='over';statRows($('overStats'));show('scrOver');
    }
    updateFx(dt);
  }else if(S.mode==='winning'){
    S.winT+=dt;
    pl.vx=0;
    if(Math.random()<.35)
      parts.push({x:totem.x-50+Math.random()*100,y:totem.y-100-Math.random()*80,
        vx:(Math.random()-.5)*220,vy:-150-Math.random()*180,g:400,t:0,life:1.2,sz:3.5,
        col:Math.random()<.5?'#3ddad7':'#ffd166'});
    updateCam(dt);updateFx(dt);
    if(S.winT>1.7){
      S.mode='win';saveBest();
      $('rankBadge').textContent=rank();
      statRows($('winStats'));show('scrWin');
    }
  }
  render();
}
